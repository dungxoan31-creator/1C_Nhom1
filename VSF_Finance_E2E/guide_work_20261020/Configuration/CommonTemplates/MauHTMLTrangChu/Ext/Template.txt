            var width = 540, height = 200, left = 48, right = 16, top = 18, bottom = 34;
            var plotHeight = height - top - bottom;
            var plotWidth = width - left - right;
            var maxValue = 0;
            rows.forEach(function(row) {
                maxValue = Math.max(maxValue, Number(row.DoanhThu) || 0, Number(row.TienThu) || 0);
            });
            if (maxValue <= 0) maxValue = 1;
            for (var grid = 0; grid <= 3; grid++) {
                var y = top + plotHeight * grid / 3;
                svg.appendChild(createSvgElement("line", {
                    x1: left, y1: y, x2: width - right, y2: y,
                    stroke: grid === 3 ? "#CBD5E1" : "#E2E8F0",
                    "stroke-dasharray": grid === 3 ? "0" : "4"
                }));
                var label = createSvgElement("text", {
                    x: left - 6, y: y + 4, fill: "#94A3B8",
                    "font-size": "9", "text-anchor": "end"
                });
                label.textContent = compactMoney(maxValue * (3 - grid) / 3);
                svg.appendChild(label);
            }
            if (rows.length === 0) {
                var empty = createSvgElement("text", {
                    x: width / 2, y: height / 2, fill: "#64748B",
                    "font-size": "12", "text-anchor": "middle"
                });
                empty.textContent = "Chua co du lieu";
                svg.appendChild(empty);
                return;
            }
            var points = [];
            var step = plotWidth / rows.length;
            rows.forEach(function(row, index) {
                var center = left + step * (index + 0.5);
                var revenue = Number(row.DoanhThu) || 0;
                var cashIn = Number(row.TienThu) || 0;
                var barHeight = plotHeight * revenue / maxValue;
                var cashY = top + plotHeight * (1 - cashIn / maxValue);
                svg.appendChild(createSvgElement("rect", {
                    x: center - Math.min(12, step * 0.22), y: top + plotHeight - barHeight,
                    width: Math.min(24, step * 0.44), height: barHeight, rx: 3, fill: "#EA0B2A"
                }));
                points.push(center + "," + cashY);
                var month = createSvgElement("text", {
                    x: center, y: height - 10, fill: "#64748B",
                    "font-size": "10", "font-weight": "600", "text-anchor": "middle"
                });
                month.textContent = row.Ky || "";
                svg.appendChild(month);
            });
            if (points.length > 1) {
                svg.appendChild(createSvgElement("polyline", {
                    points: points.join(" "), fill: "none", stroke: "#10B981",
                    "stroke-width": 3, "stroke-linecap": "round", "stroke-linejoin": "round"
                }));
            }
            rows.forEach(function(row, index) {
                var center = left + step * (index + 0.5);
                var cashY = top + plotHeight * (1 - (Number(row.TienThu) || 0) / maxValue);
                svg.appendChild(createSvgElement("circle", {
                    cx: center, cy: cashY, r: 3.5, fill: "#FFFFFF",
                    stroke: "#10B981", "stroke-width": 2
                }));
            });
        }

        function updateAgingChart(data) {
            var svgGroup = document.getElementById("ageSlices");
            var legend = document.getElementById("legendARaging");
            var buckets = Array.isArray(data.BieuDoTuoiNo) ? data.BieuDoTuoiNo : [];
            var colors = ["#10B981", "#F59E0B", "#EF4444", "#991B1B", "#7F1D1D"];
            var labels = ["Trong han", "Qua han 1-30 ngay", "Qua han 31-60 ngay", "Qua han 61-90 ngay", "Qua han tren 90 ngay"];
            var total = buckets.reduce(function(sum, item) { return sum + (Number(item.SoTien) || 0); }, 0);
            if (!svgGroup || !legend) return;
            while (svgGroup.firstChild) svgGroup.removeChild(svgGroup.firstChild);
            var circumference = 2 * Math.PI * 36;
            var offset = 0;
            buckets.forEach(function(item, index) {
                var amount = Number(item.SoTien) || 0;
                var segment = total > 0 ? circumference * amount / total : 0;
                if (segment > 0) {
                    svgGroup.appendChild(createSvgElement("circle", {
                        cx: 50, cy: 50, r: 36, fill: "transparent", stroke: colors[index],
                        "stroke-width": 16, "stroke-dasharray": segment + " " + (circumference - segment),
                        "stroke-dashoffset": -offset
                    }));
                    offset += segment;
                }
            });
            var currentAmount = buckets.length > 0 ? Number(buckets[0].SoTien) || 0 : 0;
            document.getElementById("valARInTimePct").textContent =
                (total > 0 ? Math.round(currentAmount * 100 / total) : 0) + "%";
            legend.innerHTML = buckets.map(function(item, index) {
                var amount = Number(item.SoTien) || 0;
                var percent = total > 0 ? Math.round(amount * 100 / total) : 0;
                return "<div style='display:flex;justify-content:space-between;gap:8px;align-items:center'>" +
                    "<span style='display:flex;align-items:center;gap:6px'><span style='width:8px;height:8px;border-radius:50%;background:" +
                    colors[index] + "'></span>" + labels[index] + "</span><b>" + percent + "%</b></div>" +
                    "<div style='background:#E2E8F0;height:4px;border-radius:2px;overflow:hidden;margin-top:-6px'>" +
                    "<div style='background:" + colors[index] + ";width:" + percent + "%;height:100%'></div></div>" +
                    "<div style='font-size:10px;color:#64748B;margin-top:-6px'>" + formatMoney(amount) + "</div>";
            }).join("");
        }
