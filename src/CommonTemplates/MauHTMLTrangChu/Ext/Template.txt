    <script>
        function formatMoney(amount) {
            if (!amount) return "0 VNĐ";
            return new Intl.NumberFormat('vi-VN').format(amount) + " VNĐ";
        }

        function updateDashboardData(data) {
            if (typeof data === "string") {
                try { data = JSON.parse(data); } catch(e) { return; }
            }
            if (!data) return;

            if (data.TongDoanhThu !== undefined) document.getElementById("valDoanhThu").innerText = formatMoney(data.TongDoanhThu);
            if (data.SoHoaDonBan !== undefined) document.getElementById("valSoHoaDon").innerText = data.SoHoaDonBan + " hóa đơn";
            if (data.NoPhaiThu !== undefined) document.getElementById("valPhaiThu").innerText = formatMoney(data.NoPhaiThu);
            if (data.NoPhaiThuQuaHan !== undefined) document.getElementById("valPhaiThuQuaHan").innerText = formatMoney(data.NoPhaiThuQuaHan);
            if (data.NoPhaiTra !== undefined) document.getElementById("valPhaiTra").innerText = formatMoney(data.NoPhaiTra);
            if (data.NoPhaiTraDenHan !== undefined) document.getElementById("valPhaiTraDenHan").innerText = formatMoney(data.NoPhaiTraDenHan);
            if (data.TienKhaDung !== undefined) document.getElementById("valTienKhaDung").innerText = formatMoney(data.TienKhaDung);

            // Render Top No Qua Han
            var tbodyAR = document.getElementById("tableTopNoQuaHan");
            if (data.TopNoQuaHan && data.TopNoQuaHan.length > 0) {
                var htmlAR = "";
                for (var i = 0; i < data.TopNoQuaHan.length; i++) {
                    var item = data.TopNoQuaHan[i];
                    htmlAR += "<tr>" +
                        "<td><code>" + (item.MaDoiTac || "---") + "</code></td>" +
                        "<td><b>" + (item.TenDoiTac || "Chưa xác định") + "</b></td>" +
                        "<td class='text-right font-semibold' style='color: var(--danger);'>" + formatMoney(item.SoDuNo) + "</td>" +
                        "<td class='text-right'><span class='badge-pill pill-danger'>" + (item.SoNgayQuaHan || 0) + " ngày</span></td>" +
                        "</tr>";
                }
                tbodyAR.innerHTML = htmlAR;
            } else {
                tbodyAR.innerHTML = "<tr><td colspan='4' style='text-align: center; color: var(--success);'>Tuyệt vời! Không có công nợ quá hạn.</td></tr>";
            }

            // Render Pending docs
            var tbodyDocs = document.getElementById("tableTopChungTu");
            if (data.TopChungTuChoDuyet && data.TopChungTuChoDuyet.length > 0) {
                var htmlDoc = "";
                for (var j = 0; j < data.TopChungTuChoDuyet.length; j++) {
                    var doc = data.TopChungTuChoDuyet[j];
                    htmlDoc += "<tr style='cursor:pointer;' onclick=\"window.location='vsmart://open=" + (doc.LoaiChungTu || "") + "'\">" +
                        "<td><span class='badge-pill pill-warning'>" + (doc.TenLoaiChungTu || doc.LoaiChungTu) + "</span></td>" +
                        "<td><b style='color:#EA0B2A;'>" + (doc.SoChungTu || "---") + "</b></td>" +
                        "<td class='text-right font-semibold'>" + formatMoney(doc.SoTien) + "</td>" +
                        "</tr>";
                }
                tbodyDocs.innerHTML = htmlDoc;
                document.getElementById("badgePendingCount").innerText = data.TopChungTuChoDuyet.length;
            } else {
                tbodyDocs.innerHTML = "<tr><td colspan='3' style='text-align: center; color: var(--text-muted);'>Tất cả chứng từ đã được ghi sổ</td></tr>";
                document.getElementById("badgePendingCount").innerText = "0";
            }
        }

        function refreshData() {
            // Goi nguoc len 1C Form event
            window.location = "vsmart://action=refresh";
        }
    </script>
